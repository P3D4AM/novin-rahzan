	@Val nvarchar(max)
AS
BEGIN
	SET NOCOUNT ON;

		end
	commit tran
END
Go

ALTER Procedure [dbo].[sp_Acc_CheckedSave_Save]
	@ReportType smallint,
	@EntityGuid uniqueidentifier,
	@DetailGuid uniqueIdentifier,
	@ISDebit bit,
	@Guid uniqueidentifier,
	@dateM datetime
As
Begin
	declare @count int
	set @count=(select COUNT(*) from Global_CheckedSave where [ReportType]=@ReportType and [EntityGuid]=@EntityGuid and [DetailGuid]=@DetailGuid and [ISDebit]=@ISDebit)
	set @count=ISNULL(@count,0)
	if (@count<=0)
	begin
		INSERT INTO [dbo].[Global_CheckedSave]
           ([Guid]
           ,[Modified]
           ,[IsDebit]
           ,[ReportType]
           ,[DateM]
           ,[EntityGuid]
           ,[DetailGuid])
     VALUES
           (@Guid
           ,GETDATE()
           ,@ISDebit
           ,@ReportType
           ,@dateM
           ,@EntityGuid
           ,@DetailGuid)
	end
End
Go

ALTER   PROCEDURE [dbo].[sp_Global_TicketDet_SelectAll]
	@TafsilGuid uniqueidentifier=null,
	@CashierGuid uniqueidentifier=null,
	@MasterGuid uniqueidentifier=null
AS
BEGIN
	select    Global_TicketDetail.*
			, Global_Ticket.*
			, case when Global_TicketDetail.SourceType=2 
				   then ISNULL(Acc_Cashier.Name,'')
				   else ISNULL(Acc_Tafsil.Name,'') end as SourceName
			 , ISNULL(Global_TicketPart.Name,'') As PartName
			 , ISNULL(Acc_SanadStatusReport.Name,'') As StatusName
	from Global_TicketDetail
	inner join Global_Ticket on Global_TicketDetail.MasterGuid=Global_Ticket.Guid
	left join Global_TicketPart on Global_TicketPart.Guid=Global_TicketDetail.SourceGuid
	left join Acc_SanadStatusReport on Acc_SanadStatusReport.Id=Global_Ticket.StatusId
	left outer join Acc_Cashier on Acc_Cashier.Guid=Global_TicketDetail.SourceGuid and Global_TicketDetail.SourceType=2
	left outer join Acc_Tafsil on Acc_Tafsil.Guid=Global_TicketDetail.SourceGuid and Global_TicketDetail.SourceType=1
	where (@TafsilGuid is null or @TafsilGuid = Global_Ticket.SourceGuid) and
	      (@CashierGuid is null or @CashierGuid = Global_Ticket.SourceGuid) and
		  (@MasterGuid is null or @MasterGuid = Global_TicketDetail.MasterGuid)
END
Go
